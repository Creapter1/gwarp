  (function assertDangerKeysInGallery() {
    const g = window.GW_EXT_GALLERY || [];
    const keyOf = e => {
      const slug = e.slug || '';
      if (e.src === 'TurboWarp') return 'tw:' + slug;
      if (e.src === 'PenguinMod') return 'pm:' + slug;
      return null;
    };
    const set = new Set(g.map(keyOf).filter(Boolean));
    const D = window.GW_EXT_DANGER;
    if (!D?.LIST || !D?.MODERATE) return console.warn('no GW_EXT_DANGER');
    const miss = [];
    for (const k of Object.keys(D.LIST)) if (!set.has(k)) miss.push('LIST:'+k);
    for (const k of Object.keys(D.MODERATE)) if (!set.has(k)) miss.push('MOD:'+k);
    if (miss.length) console.error('[G-Warp] DANGER keys missing from gallery', miss);
    else console.log('[G-Warp] DANGER keys ⊆ gallery OK', Object.keys(D.LIST).length, Object.keys(D.MODERATE).length);
  })();
